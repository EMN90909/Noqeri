cmake_minimum_required(VERSION 3.16)
project(noqeri_generated_c LANGUAGES C)
if(NOT DEFINED NOQERI_GENERATED_SOURCE OR NOT EXISTS "${NOQERI_GENERATED_SOURCE}")
  message(FATAL_ERROR "NOQERI_GENERATED_SOURCE must refer to emitted C source")
endif()
add_executable(noqeri_generated "${NOQERI_GENERATED_SOURCE}")
set_property(TARGET noqeri_generated PROPERTY C_STANDARD 11)
set_property(TARGET noqeri_generated PROPERTY C_STANDARD_REQUIRED ON)
if(NOT WIN32)
  target_link_libraries(noqeri_generated PRIVATE m)
endif()
